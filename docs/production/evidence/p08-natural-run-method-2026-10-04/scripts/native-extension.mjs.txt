/** Adaptive native-only operations in the same browser/run. No evaluated game writes. */
import assert from 'node:assert/strict';
import {readFile} from 'node:fs/promises';
export async function run(h){
 let command={op:'inspect'};try{command=JSON.parse(await readFile(new URL('./native-request.json',import.meta.url),'utf8'));}catch{}
 const {page,report,observe,snapshot,tapUI,shopOffers}=h;
 if(command.op==='finalize'){
  const before=await snapshot('first-Boss-final-before-image'),s=before.checkpoint.state;assert.equal(s.phase,'stage-cleared');assert.equal(s.stage.index,2);assert.equal(s.commandSeq,26);assert.equal(s.stage.heat,'1018');assert.equal(s.gold,10);assert.equal(s.stage.goldEarned,8);assert.equal(s.consumables.filter(t=>t.instanceId.endsWith('/first-boss/supply')).length,1);assert.equal(report.recoveries.length,3);
  await page.waitForFunction(()=>{const i=window.__harness.game.scene.getScene('intermission');return i.scene.isActive()&&i.ready&&i.tweens.getTweens().length===0;},null,{timeout:12000});
  const file='04-first-Boss-result.png';await page.screenshot({path:'<WORKTREE>/shots/p08-natural-run/'+file,fullPage:true});const after=await snapshot('first-Boss-final-after-image');assert.deepEqual(after.checkpoint,before.checkpoint);assert.deepEqual(after.slots,before.slots);
  report.images.push({file,terminal:true,actualResult:after.public.visibleText,checkpointBeforeEqualAfter:true});report.status='PASS';report.firstBossResult='COVERED';report.stop={phase:s.phase,stageIndex:s.stage.index,Boss:s.stage.boss,heat:s.stage.heat,target:s.stage.targetHeat,gold:s.gold,seq:s.commandSeq,journal:before.checkpoint.journal.length,clearId:s.stage.clearId,awardedGold:s.stage.goldEarned,firstBossSupply:s.consumables};report.rewardOnce={committedAndSettledCheckpointEqual:true,postResultImageCheckpointAndSlotsEqual:true,oneNewFinalPlayReceipt:true,supplyInstances:1,finalSeq:s.commandSeq,goldBeforeFinalPlay:2,goldAfterFinalPlay:s.gold,goldDelta:8};report.productChanges=false;report.tools.coverage='COVERED';report.resume=null;await h.save();return after;
 }
 if(command.op==='checkpoint'){
  const o=await snapshot('paused-parent-checkpoint');await h.image('03-Boss-paused-checkpoint');report.status='PAUSED_AT_PARENT_REQUEST';report.productChanges=false;report.tools.coverage=report.tools.used.length?'COVERED':'NOT_COVERED';report.firstBossResult='NOT_COVERED';report.resume={sameBrowser:true,PTYsession:"<REDACTED>",nodePID:"<REDACTED>",previewPort:"<REDACTED>",inspectorPort:"<REDACTED>",nativeCaller:'<TEMP_DIR>/p08-native-call.mjs',requestFile:'shots/p08-natural-run/native-request.json',nextStep:'Continue the same Boss using only current visible cards; no further reload. Stop at Boss settlement or natural failure.'};await h.save();return o;
 }
 if(command.op==='play'){
  await h.ready();await page.evaluate(()=>{if(window.__nf)return;const g=window.__harness.game;window.__nf=[];g.events.on('postrender',()=>{const s=g.scene.getScene('game');if(!s?.scene.isActive()||!s.presentation)return;const id=s.scoreTotal?.getData('eventId'),phase=s.scoreTotal?.getData('eventPhase'),previous=window.__nf.at(-1);if(!id||id===previous?.eventId&&phase===previous?.phase)return;window.__nf.push({frame:g.loop.frame,at:performance.now(),eventId:id,phase,H:s.scoreHeat?.text,M:s.scoreMult?.text,total:s.scoreTotal?.text});if(window.__nf.length>256)window.__nf.shift();});});
  await h.select(command.ids);const before=await snapshot('native-play-selected',{chosenVisibleIds:command.ids,choiceReason:command.reason});await tapUI(page,'game','action/play',true);await page.waitForFunction(seq=>window.__harness.game.registry.get('runController').state.commandSeq===seq+1,before.checkpoint.state.commandSeq);await page.waitForFunction(()=>window.__harness.game.registry.get('runController').status==='idle');const committed=await snapshot('native-play-committed');assert.equal(committed.checkpoint.journal.at(-1).action.type,'PlayHand');
  await page.waitForFunction(()=>{const g=window.__harness.game,s=g.scene.getScene('game'),i=g.scene.getScene('intermission');return i.scene.isActive()&&i.ready||s.scene.isActive()&&s.ready&&!s.playing&&!s.presentation;},null,{timeout:45000});
  const active=await observe();if(active.active.includes('game'))await h.ready();const after=await snapshot('native-play-settled');assert.deepEqual(after.checkpoint,committed.checkpoint);assert.deepEqual(after.slots,committed.slots);report.sources.push({seq:after.checkpoint.state.commandSeq,trace:after.checkpoint.state.lastTrace,actualPhaseFrames:await page.evaluate(()=>window.__nf??[])});await h.save();return after;
 }
 if(command.op==='inspect')return snapshot('inspection');
 if(command.op==='buy'){
  const kind=command.kind??'jokers';await tapUI(page,'shop','action/shelf-'+kind,true);const shelf=await shopOffers();const offer=command.definitionId?shelf.offers.find(o=>o.definitionId===command.definitionId&&!o.reason):shelf.offers.find(o=>!o.reason&&!o.consumed);
  if(!offer){await snapshot('no-affordable-'+kind,{visibleShelf:shelf});return {notCovered:'no naturally affordable requested good',shelf};}
  const before=await observe();await tapUI(page,'shop','offer/'+offer.offerId,true);const description=await page.getByRole('dialog').innerText();await page.getByRole('button',{name:'确认购买',exact:true}).tap();await page.waitForFunction(seq=>window.__harness.game.registry.get('runController').state.commandSeq===seq+1,before.checkpoint.state.commandSeq);await page.waitForFunction(()=>window.__harness.game.scene.getScene('shop').ready);const after=await snapshot('natural-bought-'+offer.definitionId,{offer,description});assert.equal(after.checkpoint.state.gold,before.checkpoint.state.gold-offer.price);
  if(kind==='tools'){assert.equal(after.checkpoint.state.consumables.length,before.checkpoint.state.consumables.length+1);report.tools.bought.push({offer,description,seq:after.checkpoint.state.commandSeq});}else if(kind==='jokers')assert.equal(after.checkpoint.state.jokers.length,before.checkpoint.state.jokers.length+1);
  await tapUI(page,'shop','action/shelf-jokers',true);await h.save();return after;
 }
 if(command.op==='open-inventory'){const active=(await observe()).active.includes('shop')?'shop':'game';await tapUI(page,active,'action/tool-inventory',true);return snapshot('natural-tool-inventory');}
 if(command.op==='tool-detail'){await page.locator('.detail-dialog[open] button').filter({hasText:command.label}).first().tap();return snapshot('natural-tool-use-detail');}
 if(command.op==='tool-select'){if(command.label)await page.getByLabel(command.label).selectOption(command.value);else await page.locator(command.selector).nth(command.index??0).tap();return snapshot('natural-tool-target-selected');}
 if(command.op==='tool-confirm'){const before=await observe();await page.getByRole('button',{name:'确认使用',exact:true}).tap();await page.waitForFunction(seq=>window.__harness.game.registry.get('runController').state.commandSeq===seq+1,before.checkpoint.state.commandSeq);await page.waitForFunction(()=>window.__harness.game.registry.get('runController').status==='idle');const after=await snapshot('natural-tool-used');assert.equal(after.checkpoint.state.consumables.length,before.checkpoint.state.consumables.length-1);report.tools.used.push({seq:after.checkpoint.state.commandSeq,action:after.checkpoint.journal.at(-1).action,before:before.checkpoint.state.consumables,after:after.checkpoint.state.consumables});await h.save();return after;}
 if(command.op==='close-dialog'){await page.getByRole('button',{name:command.label??'关闭',exact:true}).tap();return snapshot('native-dialog-closed');}
 throw Error('unknown native extension request');
}
